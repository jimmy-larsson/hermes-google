```python
_NATIVE_PREFIX = "application/vnd.google-apps."
_EXPORT_DEFAULTS = {
    "application/vnd.google-apps.document": "text/markdown",
    "application/vnd.google-apps.spreadsheet": "text/csv",
    "application/vnd.google-apps.presentation": "application/pdf",
}
_EXPORT_EXTENSIONS = {"text/markdown": ".md", "text/csv": ".csv", "application/pdf": ".pdf"}


def _export_type(mime_type: str, export_mime_type: str | None) -> str | None:
    if not mime_type.startswith(_NATIVE_PREFIX):
        return None
    if mime_type == FOLDER_MIME_TYPE:
        raise DriveError("cannot download a folder; use drive_list")
    chosen = export_mime_type or _EXPORT_DEFAULTS.get(mime_type)
    if not chosen:
        raise DriveError(f"cannot download {mime_type}: pass export_mime_type")
    return chosen


def _with_extension(name: str, export_mime_type: str) -> str:
    ext = _EXPORT_EXTENSIONS.get(export_mime_type) or mimetypes.guess_extension(export_mime_type)
    if not ext or name.lower().endswith(ext):
        return name
    return name + ext
```

```python
def get_file(
    service: Any,
    *,
    file_id: str,
    cache_dir: Path,
    export_mime_type: str | None = None,
) -> Path:
    try:
        meta = service.files().get(fileId=file_id, fields="id,name,mimeType").execute()
    except Exception as exc:  # noqa: BLE001
        raise DriveError(api_failure("get metadata", exc)) from exc

    export_as = _export_type(meta.get("mimeType", ""), export_mime_type)
    raw_name = meta["name"]
    if export_as:
        raw_name = _with_extension(raw_name, export_as)
    safe_name = os.path.basename(raw_name)
    if safe_name != raw_name or _is_unsafe_name(safe_name):
        raise DriveError(f"unsafe filename: {raw_name!r}")

    target_dir = cache_dir / "drive" / file_id
    target_dir.mkdir(parents=True, exist_ok=True)
    target = target_dir / safe_name

    # I1 (containment check): guard against symlink / absolute-path bypass.
    if not str(target.resolve()).startswith(str(target_dir.resolve()) + os.sep):
        raise DriveError(f"unsafe filename: {raw_name!r}")

    if export_as:
        request = service.files().export_media(fileId=file_id, mimeType=export_as)
    else:
        request = service.files().get_media(fileId=file_id)
    # I3: stream directly to disk — avoid buffering the full file in RAM.
    try:
        with target.open("wb") as fh:
            downloader = MediaIoBaseDownload(fh, request)
            done = False
            while not done:
                _status, done = downloader.next_chunk()
    except Exception as exc:  # noqa: BLE001
        target.unlink(missing_ok=True)
        raise DriveError(api_failure("export" if export_as else "download", exc)) from exc
    return target
```

```python
def _auto_retry(fn):
    """Retry once with fresh credentials when the stored token was replaced.

    After ``auth login`` writes a new token.json the cached credentials are
    stale and fail with AuthError; the cache is dropped and the call runs
    again against the new token.  Any other ServiceError is reported at
    once, so a request that reached Google is never repeated.
    """

    @functools.wraps(fn)
    def wrapper(*args, **kwargs):
        try:
            return fn(*args, **kwargs)
        except AuthError:
            _reset_services()
            try:
                return fn(*args, **kwargs)
            except ServiceError as exc:
                raise ToolError(str(exc)) from exc
        except ServiceError as exc:
            raise ToolError(str(exc)) from exc

    return wrapper
```

```python
"""Attachment cache hygiene: expire old downloads and wipe on disconnect."""

from __future__ import annotations

import shutil
import time
from pathlib import Path

MAX_AGE_DAYS = 30


def prune_cache(cache_dir: Path, *, max_age_days: int = MAX_AGE_DAYS, now: float | None = None) -> int:
    """Delete cached files older than ``max_age_days`` and directories left empty.

    Returns the number of files removed. A missing cache directory is fine.
    """
    if not cache_dir.is_dir():
        return 0
    cutoff = (time.time() if now is None else now) - max_age_days * 86400
    removed = 0
    for path in sorted(cache_dir.rglob("*"), key=lambda p: len(p.parts), reverse=True):
        try:
            if path.is_symlink():
                continue
            if path.is_file():
                if path.stat().st_mtime < cutoff:
                    path.unlink()
                    removed += 1
            elif path.is_dir() and not any(path.iterdir()):
                path.rmdir()
        except OSError:
            continue
    return removed


def clear_cache(cache_dir: Path) -> None:
    """Remove everything under ``cache_dir`` and keep the directory itself."""
    if not cache_dir.is_dir():
        return
    for child in cache_dir.iterdir():
        if child.is_dir() and not child.is_symlink():
            shutil.rmtree(child, ignore_errors=True)
        else:
            child.unlink(missing_ok=True)
```
